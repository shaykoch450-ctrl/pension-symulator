    <script type="text/babel">

        const { useState, useEffect, useRef } = React;


        function PensionSimulator() {

            const [salary, setSalary] = useState(15000);

            const [contribution, setContribution] = useState(18);

            const [years, setYears] = useState(35);

            const [annualYield, setAnnualYield] = useState(5);

            const [conversionFactor, setConversionFactor] = useState(200); // פרמטר חדש

            

            const chartRef = useRef(null);

            const chartInstance = useRef(null);


            const calculateData = () => {

                const guaranteedYield = 5.15;

                const monthlyRateMarket = Math.pow(1 + annualYield / 100, 1 / 12) - 1;

                const monthlyRateGuaranteed = Math.pow(1 + guaranteedYield / 100, 1 / 12) - 1;

                

                const totalMonthlyDeposit = salary * (contribution / 100);

                const depositMarket = totalMonthlyDeposit * 0.70;

                const depositGuaranteed = totalMonthlyDeposit * 0.30;

                

                const totalMonths = years * 12;

                

                let balanceMarket = 0;

                let balanceGuaranteed = 0;

                let totalInvested = 0;

                

                const dataPointsTotal = [];

                const dataPointsMarket = [];

                const dataPointsGuaranteed = [];

                const labels = [];


                for (let i = 0; i <= totalMonths; i++) {

                    if (i % 12 === 0) {

                        labels.push(`שנה ${i / 12}`);

                        dataPointsTotal.push(Math.round(balanceMarket + balanceGuaranteed));

                        dataPointsMarket.push(Math.round(balanceMarket));

                        dataPointsGuaranteed.push(Math.round(balanceGuaranteed));

                    }

                    balanceMarket = (balanceMarket + depositMarket) * (1 + monthlyRateMarket);

                    balanceGuaranteed = (balanceGuaranteed + depositGuaranteed) * (1 + monthlyRateGuaranteed);

                    totalInvested += totalMonthlyDeposit;

                }


                const finalBalance = balanceMarket + balanceGuaranteed;


                return {

                    labels,

                    dataPointsTotal,

                    dataPointsMarket,

                    dataPointsGuaranteed,

                    finalBalance,

                    totalInvested,

                    totalProfit: finalBalance - totalInvested,

                    monthlyPension: finalBalance / conversionFactor // השימוש במקדם הדינאמי

                };

            };


            const results = calculateData();


            useEffect(() => {

                const ctx = chartRef.current.getContext('2d');

                

                if (chartInstance.current) {

                    chartInstance.current.destroy();

                }


                chartInstance.current = new Chart(ctx, {

                    type: 'line',

                    data: {

                        labels: results.labels,

                        datasets: [

                            {

                                label: 'שוק ההון (70%)',

                                data: results.dataPointsMarket,

                                borderColor: '#3b82f6',

                                backgroundColor: 'rgba(59, 130, 246, 0.5)',

                                fill: true,

                                tension: 0.4,

                                pointRadius: 0

                            },

                            {

                                label: 'הבטחת תשואה 5.15% (30%)',

                                data: results.dataPointsGuaranteed,

                                borderColor: '#10b981',

                                backgroundColor: 'rgba(16, 185, 129, 0.5)',

                                fill: true,

                                tension: 0.4,

                                pointRadius: 0

                            }

                        ]

                    },

                    options: {

                        responsive: true,

                        maintainAspectRatio: false,

                        plugins: {

                            legend: { display: true, position: 'bottom' },

                            tooltip: {

                                mode: 'index',

                                intersect: false,

                                callbacks: {

                                    label: function(context) {

                                        let label = context.dataset.label || '';

                                        if (label) {

                                            label += ': ';

                                        }

                                        if (context.parsed.y !== null) {

                                            label += new Intl.NumberFormat('he-IL', { style: 'currency', currency: 'ILS', maximumFractionDigits: 0 }).format(context.parsed.y);

                                        }

                                        return label;

                                    }

                                }

                            }

                        },

                        interaction: {

                            mode: 'index',

                            intersect: false,

                        },

                        scales: {

                            y: {

                                stacked: true,

                                beginAtZero: true,

                                ticks: {

                                    callback: (value) => '₪' + (value / 1000000).toFixed(1) + 'M'

                                }

                            }

                        }

                    }

                });

            }, [salary, contribution, years, annualYield]); // עדכון הגרף קורה רק כשמשנים פרמטרים שמשפיעים על הצבירה


            const formatMoney = (num) => new Intl.NumberFormat('he-IL', { style: 'currency', currency: 'ILS', maximumFractionDigits: 0 }).format(num);


            return (

                <div className="max-w-5xl mx-auto p-4 md:p-8">

                    <header className="mb-6 text-center">

                        <h1 className="text-4xl font-bold text-gray-800 mb-2">סימולטור הפרישה האינטראקטיבי</h1>

                        <p className="text-gray-600">שחקו עם הפרמטרים כדי לראות את כוחה של הריבית דריבית</p>

                    </header>

                    

                    <div className="bg-blue-50 border-r-4 border-blue-500 p-4 rounded-lg mb-8 max-w-3xl mx-auto">

                        <p className="text-sm text-blue-800 text-center">

                            <strong>המודל הישראלי:</strong> הסימולציה מפרידה בין 30% מהכסף שזוכה להבטחת תשואה קבועה (5.15%), לבין 70% שמושקע בשוק ההון.

                        </p>

                    </div>


                    <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">

                        {/* Controls */}

                        <div className="lg:col-span-1 bg-white p-6 rounded-2xl card space-y-7">

                            <div>

                                <label className="flex justify-between text-sm font-semibold text-gray-700 mb-2">

                                    <span>שכר חודשי (ברוטו)</span>

                                    <span className="text-blue-600">{formatMoney(salary)}</span>

                                </label>

                                <input type="range" min="5000" max="50000" step="500" value={salary} onChange={(e) => setSalary(Number(e.target.value))} className="w-full h-2 bg-gray-200 rounded-lg appearance-none cursor-pointer slider-thumb" />

                            </div>


                            <div>

                                <label className="flex justify-between text-sm font-semibold text-gray-700 mb-2">

                                    <span>שיעור הפרשה (%)</span>

                                    <span className="text-blue-600">{contribution}%</span>

                                </label>

                                <input type="range" min="10" max="25" step="0.5" value={contribution} onChange={(e) => setContribution(Number(e.target.value))} className="w-full h-2 bg-gray-200 rounded-lg appearance-none cursor-pointer slider-thumb" />

                            </div>


                            <div>

                                <label className="flex justify-between text-sm font-semibold text-gray-700 mb-2">

                                    <span>משך הפרשה (שנים)</span>

                                    <span className="text-blue-600">{years} שנים</span>

                                </label>

                                <input type="range" min="10" max="50" step="1" value={years} onChange={(e) => setYears(Number(e.target.value))} className="w-full h-2 bg-gray-200 rounded-lg appearance-none cursor-pointer slider-thumb" />

                            </div>


                            <div>

                                <label className="flex justify-between text-sm font-semibold text-gray-700 mb-2">

                                    <span>תשואת שוק שנתית (על 70%)</span>

                                    <span className="text-blue-600">{annualYield}%</span>

                                </label>

                                <input type="range" min="1" max="15" step="0.1" value={annualYield} onChange={(e) => setAnnualYield(Number(e.target.value))} className="w-full h-2 bg-gray-200 rounded-lg appearance-none cursor-pointer slider-thumb" />

                            </div>


                            <div className="pt-4 border-t border-gray-200">

                                <label className="flex justify-between text-sm font-semibold text-gray-700 mb-2">

                                    <span>מקדם המרה פנסיוני</span>

                                    <span className="text-emerald-600">{conversionFactor}</span>

                                </label>

                                <input type="range" min="150" max="250" step="1" value={conversionFactor} onChange={(e) => setConversionFactor(Number(e.target.value))} className="w-full h-2 bg-gray-200 rounded-lg appearance-none cursor-pointer slider-thumb" style={{background: '#e5e7eb'}} />

                                <p className="text-xs text-gray-500 mt-1">משפיע ישירות על הקצבה (ככל שהמקדם נמוך יותר, הקצבה תהיה גבוהה יותר)</p>

                            </div>

                        </div>


                        {/* Visualization & Results */}

                        <div className="lg:col-span-2 space-y-6">

                            <div className="grid grid-cols-1 md:grid-cols-3 gap-4">

                                <div className="bg-blue-600 p-4 rounded-xl text-white shadow-lg">

                                    <p className="text-xs opacity-80 mb-1">סכום צבור בפרישה</p>

                                    <p className="text-2xl font-bold">{formatMoney(results.finalBalance)}</p>

                                </div>

                                <div className="bg-emerald-500 p-4 rounded-xl text-white shadow-lg">

                                    <p className="text-xs opacity-80 mb-1">קצבה חודשית משוערת</p>

                                    <p className="text-2xl font-bold">{formatMoney(results.monthlyPension)}</p>

                                </div>

                                <div className="bg-amber-500 p-4 rounded-xl text-white shadow-lg">

                                    <p className="text-xs opacity-80 mb-1">מתוכם רווחי תשואה</p>

                                    <p className="text-2xl font-bold">{formatMoney(results.totalProfit)}</p>

                                </div>

                            </div>


                            <div className="bg-white p-6 rounded-2xl card h-80">

                                <canvas ref={chartRef}></canvas>

                            </div>


                            <div className="bg-gray-100 p-4 rounded-xl text-sm text-gray-600 italic text-center">

                                * הסימולציה אינה כוללת דמי ניהול ומיסוי ונועדה להמחשה בלבד.

                            </div>

                        </div>

                    </div>

                </div>

            );

        }


        const root = ReactDOM.createRoot(document.getElementById('root'));

        root.render(<PensionSimulator />);
